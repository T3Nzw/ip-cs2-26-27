# Задачи

## Задача 01

Какво ще изведат следните кодови фрагменти?

1) symbols as numbers
```cpp
#include <iostream>
int main() {
  std::cout << (char)104 << (char)105 << (char)33;
  return 0;
}
```

2) uppercase to lowercase
```cpp
#include <iostream>
int main() {
  char letter = 'K';
  letter += 'a' - 'A';
  std::cout << letter;
  return 0;
}
```

3) unsigned to signed conversion
```cpp
#include <iostream>
int main() {
  unsigned char a = 255;
  char b = a;
  std::cout << (int)b;
  return 0;
}
```

4) signed to unsigned conversion
```cpp
#include <iostream>
int main() {
  int num = -1;
  unsigned int unsignedNum = num;
  std::cout << unsignedNum << '\n';
  std::cout << unsignedNum+1;
  return 0;
}
```

5) floating point arithmetic
```cpp
#include <iostream>
int main() {
  float a = 0.2;
  float b = 0.1;
  if (a+b == 0.3) {
    std::cout << "computers can do simple math, of course!";
  }
  return 0;
}
```

6) assignment != comparison
```cpp
#include <iostream>
int main() {
  int a = 10;
  if (a = 1) {
    std::cout << a;
  }
  return 0;
}
```

7) forbidden arithmetic
```cpp
#include <iostream>
int main() {
  int a = 42;
  std::cout << (a == 42 && a/0);
  return 0;
}
```

8) properties of boolean operators
```cpp
#include <iostream>
int main() {
  int a = 42;
  std::cout << (a == 42 || a/0);
  std::cout << (a == 41 && a/0);
  return 0;
}
```

9) numbers can be boolean values
```cpp
#include <iostream>
int main() {
  int a = 10, b = 0;
  if (!a) {
    std::cout << "a is not zero";
  }
  else if (!b) {
    std::cout << "b is zero";
  }
  return 0;
}
```

10) mutability of variables
```cpp
#include <iostream>
int main() {
  int a = 15;
  int b = (a -= 5) * (a += 4);
  std::cout << b;
  return 0;
}
```
